set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -stdlib=libc++")
set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0")

# The SDK include path is not hardcoded here: CMake resolves the active SDK
# via CMAKE_OSX_SYSROOT (xcrun --show-sdk-path), which works with both Xcode
# and the Command Line Tools and follows SDK upgrades.  A pinned path such as
# .../MacOSX15.2.sdk/usr/include breaks on any machine with a different SDK,
# and is worse when it exists but is stale, since it mixes headers from one
# SDK with the current toolchain.  Override CMAKE_OSX_SYSROOT on the command
# line to build against a specific SDK.
add_definitions(-DPLATFORM_MACOS)

set(PLATFORM_LINK_LIBRARIES objc "-framework Foundation" "-framework IOKit" "-framework Security")
